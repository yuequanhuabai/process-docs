package com.ha.pr.flowwork.dto.response;

import com.fasterxml.jackson.annotation.JsonFormat;
import io.swagger.v3.oas.annotations.media.Schema;
import lombok.Data;
import org.flowable.engine.history.HistoricProcessInstance;
import org.flowable.engine.runtime.ProcessInstance;

import java.util.Date;

@Data
@Schema(description = "流程实例（运行中 + 历史通用）")
public class ProcessInstanceVO {

    // status 的三个取值。Flowable 不存状态列，这是按判据算出来的（见 statusOf）。
    // 用常量不用裸字符串：Service 的过滤分支、enrich 的筛选、前端契约都引用它，散写迟早拼错
    public static final String STATUS_RUNNING = "RUNNING";
    public static final String STATUS_COMPLETED = "COMPLETED";
    public static final String STATUS_TERMINATED = "TERMINATED";

    @Schema(description = "流程实例 id（引擎生成，天生字符串）")
    private String id;

    @Schema(description = "所用流程定义 id（发起时锁定的具体版本）")
    private String processDefinitionId;

    @Schema(description = "流程定义 key")
    private String processDefinitionKey;

    @Schema(description = "流程定义名称")
    private String processDefinitionName;

    @Schema(description = "业务键（关联业务表记录，可能为 null）")
    private String businessKey;

    @Schema(description = "发起人 userId（来自 Authentication，对应 START_USER_ID_）")
    private String startUserId;

    @Schema(description = "发起时间")
    @JsonFormat(pattern = "yyyy-MM-dd HH:mm:ss", timezone = "GMT+8")
    private Date startTime;

    @Schema(description = "是否挂起（仅 RU 来源有意义；HI 无挂起概念，恒 false）")
    private boolean suspended;

    @Schema(description = "结束时间（在途实例为 null）")
    @JsonFormat(pattern = "yyyy-MM-dd HH:mm:ss", timezone = "GMT+8")
    private Date endTime;

    @Schema(description = "流程状态：RUNNING / COMPLETED / TERMINATED（按判据算出，不是引擎字段）")
    private String status;

    @Schema(description = "终止原因（DELETE_REASON_；仅 TERMINATED 有值）")
    private String deleteReason;

    @Schema(description = "发起人姓名（enrich 批量回填）")
    private String startUserName;

    @Schema(description = "流程当前停留节点名（仅 RUNNING 有值，enrich 批量回填）")
    private String currentActivityName;

    /** RU 来源（page / start 用）：RU 表里的必然在途，status 直接定死 */
    public static ProcessInstanceVO of(ProcessInstance instance) {
        ProcessInstanceVO vo = new ProcessInstanceVO();
        vo.setId(instance.getId());
        vo.setProcessDefinitionId(instance.getProcessDefinitionId());
        vo.setProcessDefinitionKey(instance.getProcessDefinitionKey());
        vo.setProcessDefinitionName(instance.getProcessDefinitionName());
        vo.setBusinessKey(instance.getBusinessKey());
        vo.setStartUserId(instance.getStartUserId());
        vo.setStartTime(instance.getStartTime());
        vo.setSuspended(instance.isSuspended());
        // 不是装饰：enrich() 靠它判断"该不该查当前任务"，page() 也走 enrich
        vo.setStatus(STATUS_RUNNING);
        return vo;
    }

    /**
     * HI 来源（Day 16 myStarted 用）：全集 = 在途 + 办结 + 终止。
     * 与上面是两个不同接口（runtime 包 vs history 包），传错编译不过。
     * HistoricProcessInstance 没有 isSuspended()——挂起是纯运行时概念，suspended 留默认 false
     */
    public static ProcessInstanceVO of(HistoricProcessInstance hi) {
        ProcessInstanceVO vo = new ProcessInstanceVO();
        vo.setId(hi.getId());
        vo.setProcessDefinitionId(hi.getProcessDefinitionId());
        vo.setProcessDefinitionKey(hi.getProcessDefinitionKey());
        vo.setProcessDefinitionName(hi.getProcessDefinitionName());
        vo.setBusinessKey(hi.getBusinessKey());
        vo.setStartUserId(hi.getStartUserId());
        vo.setStartTime(hi.getStartTime());
        vo.setEndTime(hi.getEndTime());
        vo.setDeleteReason(hi.getDeleteReason());
        vo.setStatus(statusOf(hi.getEndTime(), hi.getDeleteReason()));
        return vo;
    }

    /**
     * 结局判据（hi_tables/03 第五节 = 07 文档第三节，同一张表）：
     * END_TIME_ 为 null → 在途；有值且 DELETE_REASON_ 为 null → 正常办结；两者都有值 → 被终止。
     * ⚠️ 与 ServiceImpl.applyStatusFilter 是同一套规则的两面（这边查回来打标签，那边查之前拼 WHERE），改一面必看另一面
     */
    private static String statusOf(Date endTime, String deleteReason) {
        if (endTime == null) {
            return STATUS_RUNNING;
        }
        return deleteReason == null ? STATUS_COMPLETED : STATUS_TERMINATED;
    }
}

## 3. `service/impl/ProcessInstanceServiceImpl.java`（完整替换）

> `start()` / `terminate()` 原样未动；`page()` 本体未动但因 `toPage()` 收编了 `enrich()`，自动获得两个新字段。

```java
package com.ha.pr.flowwork.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.core.metadata.IPage;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.ha.pr.flowwork.common.BizException;
import com.ha.pr.flowwork.dto.request.StartProcessInstanceRequest;
import com.ha.pr.flowwork.dto.response.ProcessInstanceVO;
import com.ha.pr.flowwork.entity.SysUser;
import com.ha.pr.flowwork.mapper.SystemUserMapper;
import com.ha.pr.flowwork.service.ProcessInstanceService;
import com.ha.pr.flowwork.util.UserContext;
import jakarta.annotation.Resource;
import lombok.extern.slf4j.Slf4j;
import org.flowable.common.engine.api.query.Query;
import org.flowable.common.engine.impl.identity.Authentication;
import org.flowable.engine.HistoryService;
import org.flowable.engine.RepositoryService;
import org.flowable.engine.RuntimeService;
import org.flowable.engine.TaskService;
import org.flowable.engine.history.HistoricProcessInstanceQuery;
import org.flowable.engine.repository.ProcessDefinition;
import org.flowable.engine.runtime.ProcessInstance;
import org.flowable.engine.runtime.ProcessInstanceQuery;
import org.flowable.task.api.Task;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Set;
import java.util.function.Function;
import java.util.stream.Collectors;

@Service
@Slf4j
public class ProcessInstanceServiceImpl implements ProcessInstanceService {
    /** 运行域门面：实例的发起/查询/终止 */
    @Resource
    private RuntimeService runtimeService;

    /** 仓库域门面：发起前校验流程定义（一个 Service 组合多个门面很正常） */
    @Resource
    private RepositoryService repositoryService;

    /** 历史域门面：ACT_HI_PROCINST 的只读查询（Day 16 myStarted 的新数据源） */
    @Resource
    private HistoryService historyService;

    /** 任务域门面：enrich 里批量查在途实例的当前任务节点 */
    @Resource
    private TaskService taskService;

    /** enrich 里批量把 startUserId 换成 sys_user.real_name */
    @Resource
    private SystemUserMapper systemUserMapper;

    @Override
    public ProcessInstanceVO start(StartProcessInstanceRequest req) {
        String key = req.getProcessDefinitionKey();
        // 预检：把引擎的生硬异常（找不到定义/定义挂起）换成统一 R 友好报错
        ProcessDefinition def = repositoryService.createProcessDefinitionQuery()
                .processDefinitionKey(key)
                .latestVersion()
                .singleResult();
        if (def == null) {
            throw new BizException("流程定义不存在：" + key);
        }
        if (def.isSuspended()) {
            throw new BizException("流程定义已挂起，无法发起：" + key);
        }
        // 接线：把我们的 UserContext 用户告诉引擎的 ThreadLocal，
        // 引擎发起时写 START_USER_ID_ + initiator 变量；用完必须清（同 UserContext 的纪律）
        Authentication.setAuthenticatedUserId(UserContext.userId());
        try {
            ProcessInstance instance = runtimeService.startProcessInstanceByKey(
                    key, req.getBusinessKey(), req.getVariables());
            log.info("流程实例已发起：instanceId={}, definitionId={}, startUserId={}",
                    instance.getId(), instance.getProcessDefinitionId(), instance.getStartUserId());
            return ProcessInstanceVO.of(instance);
        } finally {
            Authentication.setAuthenticatedUserId(null);
        }
    }

    @Override
    public IPage<ProcessInstanceVO> page(long current, long size, String definitionKey) {
        // 继续查 RU 是故意的：这一页的语义就是"现在有哪些实例在跑"，别改成 HI（07 文档）
        ProcessInstanceQuery query = runtimeService.createProcessInstanceQuery()
                .orderByStartTime().desc();
        if (definitionKey != null && !definitionKey.isBlank()) {
            query.processDefinitionKey(definitionKey);
        }
        return  (query, ProcessInstanceVO::of, current, size);
    }

    @Override
    public IPage<ProcessInstanceVO> myStarted(long current, long size, String status) {
        // Day 16 改造：RuntimeService（只见在途）→ HistoryService（全集），原理见 07 文档。
        // ⚠️ HI 的排序方法叫 orderByProcessInstanceStartTime，与 RU 的 orderByStartTime 不同名
        HistoricProcessInstanceQuery query = historyService.createHistoricProcessInstanceQuery()
                .startedBy(UserContext.userId())   // 过滤 ACT_HI_PROCINST.START_USER_ID_
                .orderByProcessInstanceStartTime().desc();
        applyStatusFilter(query, status);
        return toPage(query, ProcessInstanceVO::of, current, size);
    }

    @Override
    public void terminate(String instanceId, String reason) {
        ProcessInstance instance = runtimeService.createProcessInstanceQuery()
                .processInstanceId(instanceId)
                .singleResult();
        if (instance == null) {
            // 已完结的实例不在 RU 表里，报"不存在"是正确语义（历史在 Day 14 查）
            throw new BizException("运行中的流程实例不存在：" + instanceId);
        }
        runtimeService.deleteProcessInstance(instanceId,
                reason != null && !reason.isBlank() ? reason : "手动终止");
    }

    /**
     * 状态过滤（前端"全部/进行中/已办结/已终止"下拉）。筛选必须在后端做——
     * 放前端只能过滤当前页，total 还是全量、翻页会出空页。
     * ⚠️ 与 ProcessInstanceVO.statusOf 是同一套判据的两面（那边打标签，这边拼 WHERE），改一面必看另一面
     */
    private void applyStatusFilter(HistoricProcessInstanceQuery query, String status) {
        if (status == null || status.isBlank()) {
            return;   // 不传 = 全部
        }
        switch (status) {
            case ProcessInstanceVO.STATUS_RUNNING -> query.unfinished();               // END_TIME_ IS NULL
            case ProcessInstanceVO.STATUS_COMPLETED -> query.finished().notDeleted();  // 结束且 DELETE_REASON_ IS NULL
            case ProcessInstanceVO.STATUS_TERMINATED -> query.deleted();               // DELETE_REASON_ IS NOT NULL（终止必已结束）
            // 传错参数要响亮地报错，安静返回全量是要 debug 半天的软故障
            default -> throw new BizException("未知的状态过滤值：" + status);
        }
    }

    /**
     * 批量回填 startUserName + currentActivityName。整页固定 2 次查询，
     * 严禁在循环里逐条查（N+1：一页 10 条 = 20 次库访问）。
     * Day 17 待办页、Day 18 历史页直接复用这两个字段
     */
    private void enrich(List<ProcessInstanceVO> vos) {
        if (vos.isEmpty()) {
            return;
        }
        // ① 发起人姓名：START_USER_ID_ 存的是 sys_user.id（Long 雪花）的字符串形态，
        //    查库前 Long.valueOf 转过去，转 Map 时 String.valueOf 转回来，两头都要转
        Set<Long> userIds = vos.stream()
                .map(ProcessInstanceVO::getStartUserId)
                .filter(Objects::nonNull)
                .map(Long::valueOf)
                .collect(Collectors.toSet());
        // MP 的 in() 遇空集合会拼出非法 SQL（IN ()），必须判空短路
        Map<String, String> nameMap = userIds.isEmpty() ? Map.of()
                : systemUserMapper.selectList(
                        new LambdaQueryWrapper<SysUser>().in(SysUser::getId, userIds))
                    .stream()
                    .collect(Collectors.toMap(u -> String.valueOf(u.getId()), SysUser::getRealName));
        // ② 当前节点名：只有在途实例才有"当前任务"（已结束的不在 ACT_RU_TASK，留 null）
        List<String> runningIds = vos.stream()
                .filter(v -> ProcessInstanceVO.STATUS_RUNNING.equals(v.getStatus()))
                .map(ProcessInstanceVO::getId)
                .toList();
        // 并行网关下一个实例可能同时有多个任务，toMap 无 merge 撞 key 直接抛异常；
        // 当前四个模板都是单路径取谁都一样，(a, b) -> a 是防未来的兜底
        Map<String, String> activityMap = runningIds.isEmpty() ? Map.of()
                : taskService.createTaskQuery().processInstanceIdIn(runningIds).list()
                    .stream()
                    .collect(Collectors.toMap(Task::getProcessInstanceId, Task::getName, (a, b) -> a));
        for (ProcessInstanceVO vo : vos) {
            vo.setStartUserName(nameMap.get(vo.getStartUserId()));
            vo.setCurrentActivityName(activityMap.get(vo.getId()));
        }
    }

    /**
     * count() + listPage() 装进 MP Page 外壳（Day 11/12 套路），Day 16 泛型化：
     * RU 与 HI 两种 Query 都继承 org.flowable.common.engine.api.query.Query，
     * count/listPage 就定义在公共接口上，E 由传入的 query 推断，
     * ProcessInstanceVO::of 由编译器按 E 自动选中正确的重载。
     * enrich 收在这里，page() 与 myStarted() 都自动获得回填
     */
    private <E> IPage<ProcessInstanceVO> toPage(Query<?, E> query,
                                                Function<E, ProcessInstanceVO> mapper,
                                                long current, long size) {
        long total = query.count();
        List<E> list = query.listPage((int) ((current - 1) * size), (int) size);
        Page<ProcessInstanceVO> page = new Page<>(current, size, total);
        page.setRecords(list.stream().map(mapper).toList());
        enrich(page.getRecords());
        return page;
    }
}
```

> **泛型版看不惯的替代写法**：保留原来吃 `ProcessInstanceQuery` 的 `toPage`，另复制一份 `toHiPage(HistoricProcessInstanceQuery query, long current, long size)` 参数写死，两个方法末尾都调 `enrich()`。功能完全等价，只是分页收尾的代码存在两份。先跑通，想收编时再回来换泛型版。
