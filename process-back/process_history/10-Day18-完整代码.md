package com.ha.pr.flowwork.dto.response;

import com.fasterxml.jackson.annotation.JsonFormat;
import io.swagger.v3.oas.annotations.media.Schema;
import lombok.Data;
import org.flowable.task.api.history.HistoricTaskInstance;

import java.util.Date;

@Data
@Schema(description = "已审批任务（历史）")
public class ApprovedTaskVO {

    @Schema(description = "任务 id")
    private String id;

    @Schema(description = "任务名称（BPMN userTask 的 name）")
    private String name;

    @Schema(description = "任务节点 key（BPMN userTask 的 id）")
    private String taskDefinitionKey;

    @Schema(description = "办理人 userId（ACT_HI_TASKINST.ASSIGNEE_，Day 13 自动认领写入）")
    private String assignee;

    @Schema(description = "所属流程实例 id")
    private String processInstanceId;

    @Schema(description = "所用流程定义 id（含版本）")
    private String processDefinitionId;

    @Schema(description = "任务创建时间")
    @JsonFormat(pattern = "yyyy-MM-dd HH:mm:ss", timezone = "GMT+8")
    private Date startTime;

    @Schema(description = "办结时间")
    @JsonFormat(pattern = "yyyy-MM-dd HH:mm:ss", timezone = "GMT+8")
    private Date endTime;

    @Schema(description = "耗时（毫秒；前端自行换算成可读时长）")
    private Long durationInMillis;

    @Schema(description = "删除原因：正常办结为 null，任务因实例终止而结束时有值")
    private String deleteReason;

    // ===== 以下 3 个字段 ACT_HI_TASKINST 上没有，由 FlowableHistoryServiceImpl.enrich()
    //       从 ACT_HI_PROCINST（不是 ACT_RU_EXECUTION！实例多半已办结）批量回填。
    //       所以 of() 里不填它们——在工厂方法里逐条查实例就是 N+1 的第一步。 =====

    @Schema(description = "流程名称（如「采购流程」，来自所属实例；enrich 回填）")
    private String processDefinitionName;

    @Schema(description = "业务标识（发起时选填，没填则为 null；enrich 回填）")
    private String businessKey;

    @Schema(description = "发起人姓名（sys_user.real_name，空则回退 username；enrich 回填）")
    private String startUserName;

    public static ApprovedTaskVO of(HistoricTaskInstance task) {
        ApprovedTaskVO vo = new ApprovedTaskVO();
        vo.setId(task.getId());
        vo.setName(task.getName());
        vo.setTaskDefinitionKey(task.getTaskDefinitionKey());
        vo.setAssignee(task.getAssignee());
        vo.setProcessInstanceId(task.getProcessInstanceId());
        vo.setProcessDefinitionId(task.getProcessDefinitionId());
        vo.setStartTime(task.getStartTime());
        vo.setEndTime(task.getEndTime());
        vo.setDurationInMillis(task.getDurationInMillis());
        vo.setDeleteReason(task.getDeleteReason());
        return vo;
    }
}

## 3. `service/impl/FlowableHistoryServiceImpl.java`（完整替换）

```java
package com.ha.pr.flowwork.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.core.metadata.IPage;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.ha.pr.flowwork.dto.response.ActivityTraceVO;
import com.ha.pr.flowwork.dto.response.ApprovedTaskVO;
import com.ha.pr.flowwork.entity.SysUser;
import com.ha.pr.flowwork.mapper.SystemUserMapper;
import com.ha.pr.flowwork.service.FlowableHistoryService;
import com.ha.pr.flowwork.util.UserContext;
import jakarta.annotation.Resource;
import org.flowable.engine.HistoryService;
import org.flowable.engine.history.HistoricActivityInstance;
import org.flowable.engine.history.HistoricProcessInstance;
import org.flowable.task.api.history.HistoricTaskInstance;
import org.flowable.task.api.history.HistoricTaskInstanceQuery;
import org.springframework.stereotype.Service;

import java.util.Comparator;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Set;
import java.util.stream.Collectors;

@Service
public class FlowableHistoryServiceImpl implements FlowableHistoryService {

    /** 历史域门面：ACT_HI_* 的只读查询 */
    @Resource
    private HistoryService historyService;

    /** 批量把 userId 换成姓名（enrich 用，避免 N+1） */
    @Resource
    private SystemUserMapper systemUserMapper;

    @Override
    public IPage<ApprovedTaskVO> approved(long current, long size) {
        // taskAssignee 查 ACT_HI_TASKINST.ASSIGNEE_（Day 13 自动认领写入的值）；
        // finished() 排除"正等我办、还没办完"的行（HI 边跑边写，未完成任务 END_TIME_ 为 null）
        HistoricTaskInstanceQuery query = historyService.createHistoricTaskInstanceQuery()
                .taskAssignee(UserContext.userId())
                .finished()
                .orderByHistoricTaskInstanceEndTime().desc();
        long total = query.count();
        List<HistoricTaskInstance> tasks = query.listPage((int) ((current - 1) * size), (int) size);
        Page<ApprovedTaskVO> page = new Page<>(current, size, total);
        page.setRecords(tasks.stream().map(ApprovedTaskVO::of).toList());
        // ⚠️ 漏掉这一行不会编译报错，只会让三个新字段整页全空（Day 17 踩过同款）
        enrich(page.getRecords());
        return page;
    }

    @Override
    public List<ActivityTraceVO> activities(String instanceId) {
        // 按进入时间升序 = 真实流转顺序；不加 finished()，好让"当前停留节点"（END_TIME_ 为 null）也出现
        List<HistoricActivityInstance> list = historyService.createHistoricActivityInstanceQuery()
                .processInstanceId(instanceId)
                .orderByHistoricActivityInstanceStartTime().asc()
                .list();
        // ⚠️ START_TIME_ 毫秒级会撞车：同事务内 <1ms 走完的多个节点（如 flow1/managerApprove）时间戳相同，
        // 仅按 startTime 排序顺序不保证且不报错。引擎有 TRANSACTION_ORDER_ 列记录同事务内先后，但
        // HistoricActivityInstanceQuery 无对应 orderBy 方法（已查 jar 确认），只能 Java 侧二次排序兜底。
        // transactionOrder 可能为 null（旧数据/历史级别不足），nullsLast 置末。
        List<ActivityTraceVO> vos = list.stream()
                .sorted(Comparator.comparing(HistoricActivityInstance::getStartTime)
                        .thenComparing(HistoricActivityInstance::getTransactionOrder,
                                Comparator.nullsLast(Comparator.naturalOrder())))
                .map(ActivityTraceVO::of)
                .toList();
        // 填谁就返回谁：别在 return 表达式里重新 map 一遍（Day 17 的 detail() 就是这么把 enrich 结果弄丢的）
        enrichAssigneeName(vos);
        return vos;
    }

    /**
     * 已审批列表批量回填：流程名 / businessKey / 发起人姓名。整页固定 2 次查询。
     *
     * <p>★ 与 Day 17 {@code FlowableTaskServiceImpl.enrich()} 的唯一但致命的差别：
     * <b>这里查 HistoryService，不是 RuntimeService</b>。已审批任务所属的实例大多已办结，
     * 那一行早已从 ACT_RU_EXECUTION 删掉，用 RuntimeService 查会一条也查不到，
     * 表现为"编译全绿、三个字段整页 null"。
     */
    private void enrich(List<ApprovedTaskVO> vos) {
        if (vos.isEmpty()) {
            // 不只是省事：下面的 processInstanceIds() 吃不得空集合（引擎直接抛异常）
            return;
        }
        // ① 任务 → 所属历史实例（1 次）。注意参数是 Set<String>，不是 List（javap 查过 7.1.0 的签名）
        Set<String> instanceIds = vos.stream()
                .map(ApprovedTaskVO::getProcessInstanceId)
                .collect(Collectors.toSet());
        Map<String, HistoricProcessInstance> instanceMap = historyService.createHistoricProcessInstanceQuery()
                .processInstanceIds(instanceIds)
                .list()
                .stream()
                .collect(Collectors.toMap(HistoricProcessInstance::getId, i -> i));

        // ② 发起人姓名（1 次）
        Set<Long> userIds = instanceMap.values().stream()
                .map(HistoricProcessInstance::getStartUserId)
                .filter(Objects::nonNull)
                .map(Long::valueOf)
                .collect(Collectors.toSet());
        Map<String, String> nameMap = nameMap(userIds);

        // ③ 回填
        for (ApprovedTaskVO vo : vos) {
            HistoricProcessInstance hi = instanceMap.get(vo.getProcessInstanceId());
            if (null == hi) {
                // 理论上不会发生（HI 任务的实例必在 HI），防御一手，免得引擎行为变化时整页 NPE
                continue;
            }
            vo.setProcessDefinitionName(hi.getProcessDefinitionName());
            vo.setBusinessKey(hi.getBusinessKey());
            vo.setStartUserName(nameMap.get(hi.getStartUserId()));
        }
    }

    /**
     * 轨迹批量回填办理人姓名。整页固定 1 次查询——轨迹本来就同属一个实例，不需要查实例。
     */
    private void enrichAssigneeName(List<ActivityTraceVO> vos) {
        if (vos.isEmpty()) {
            return;
        }
        Set<Long> userIds = vos.stream()
                .map(ActivityTraceVO::getAssignee)
                .filter(Objects::nonNull)
                // ⚠️ 本次唯一一处 Day 16/17 没有的新防御，别省：
                // START_USER_ID_ 的唯一写入口是 start() 里的 setAuthenticatedUserId(userId)，恒为数字；
                // 而 ASSIGNEE_ 还有第二个写入口——BPMN 里静态写死的 flowable:assignee="具体用户名"
                //（四个模板的注释里就明写着这条改法，见 leave.bpmn20.xml:7）。
                // 真那么改一次，这里的 Long.valueOf 会 NumberFormatException，整个轨迹接口 500。
                // 过滤掉非数字后，最坏结果只是 assigneeName 为 null，前端回退显示原 assignee。
                .filter(s -> s.chars().allMatch(Character::isDigit))
                .map(Long::valueOf)
                .collect(Collectors.toSet());
        Map<String, String> nameMap = nameMap(userIds);

        for (ActivityTraceVO vo : vos) {
            if (null != vo.getAssignee()) {
                // 取不到就是 null（事件/网关行、或上面被过滤掉的非数字 assignee），正常
                vo.setAssigneeName(nameMap.get(vo.getAssignee()));
            }
        }
    }

    /**
     * userId 集合 → Map&lt;userId 字符串, 姓名&gt;。一次查询，三个坑都在这里：
     * <ol>
     *   <li>ACT_HI_* 里的 userId 是 Long 的字符串形态：查库前 Long.valueOf 转过去，
     *       转 Map 时 String.valueOf 转回来，<b>两头都要转</b></li>
     *   <li>MP 的 in() 遇空集合会拼出非法 SQL（IN ()），必须判空短路</li>
     *   <li>Collectors.toMap 内部走 HashMap.merge，<b>value 为 null 直接抛 NPE</b>
     *       （不是存个 null 进去），而 sys_user.real_name 列可空 —— 必须兜底成 username</li>
     * </ol>
     *
     * <p>本方法与 Day 16 {@code ProcessInstanceServiceImpl.enrich()} ①段、
     * Day 17 {@code FlowableTaskServiceImpl.enrich()} ②段逐行重复，属<b>已知且暂时接受</b>的重复。
     * 跨类收编成公共的姓名服务已登记为 P4 收尾评估项，清单见 08 文档第五节。
     */
    private Map<String, String> nameMap(Set<Long> userIds) {
        if (userIds.isEmpty()) {
            return Map.of();
        }
        return systemUserMapper.selectList(
                        new LambdaQueryWrapper<SysUser>().in(SysUser::getId, userIds))
                .stream()
                .collect(Collectors.toMap(
                        u -> String.valueOf(u.getId()),
                        u -> u.getRealName() == null ? u.getUsername() : u.getRealName()
                ));
    }
}
```
