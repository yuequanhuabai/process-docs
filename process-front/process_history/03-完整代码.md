import request from '@/utils/request'
import type {PageResult} from '@/api/user'

// 对齐后端 ApprovedTaskVO（Day 18 补了最后三个字段）
export interface ApprovedTask {
    id: string
    name: string
    taskDefinitionKey: string
    assignee: string | null
    processInstanceId: string
    processDefinitionId: string
    startTime: string
    endTime: string
    durationInMillis: number | null
    deleteReason: string | null          // 正常办结为 null；非 null = 实例被终止带走的
    processDefinitionName: string | null // ↓ 三个都是 enrich() 回填的，后端没改完就是 null
    businessKey: string | null
    startUserName: string | null
}

// 对齐后端 ActivityTraceVO（Day 18 补了 assigneeName）
export interface ActivityTrace {
    activityId: string
    activityName: string | null   // startEvent/endEvent 常无 name
    activityType: string          // startEvent / userTask / endEvent / sequenceFlow …
    assignee: string | null       // userId，仅 userTask 有值
    assigneeName: string | null   // 姓名，同上
    startTime: string
    endTime: string | null        // null = 流程当前正停在此节点
    durationInMillis: number | null
}

export function pageApproved(current = 1, size = 10) {
    return request.get<never, PageResult<ApprovedTask>>('/history/approved', {
        params: {current, size}
    })
}

// ⚠️ 与前三页不同：后端返回的是【数组】，不是 PageResult
export function getActivities(instanceId: string) {
    return request.get<never, ActivityTrace[]>(`/history/instance/${instanceId}/activities`)
}

## 2. `src/utils/format.ts`（新建）

```ts
/**
 * 毫秒 → 人类可读时长。后端给的 durationInMillis 是整数（如 116000）。
 * null/undefined = 该节点还没结束（当前停留中），回退占位符。
 */
export function formatDuration(ms: number | null | undefined): string {
    if (ms === null || ms === undefined) {
        return '—'
    }
    if (ms < 1000) {
        // 真会出现：同事务内 <1ms 走完的节点
        return '不到 1 秒'
    }
    const sec = Math.floor(ms / 1000)
    if (sec < 60) {
        return `${sec} 秒`
    }
    const min = Math.floor(sec / 60)
    if (min < 60) {
        return `${min} 分钟`
    }
    const hour = Math.floor(min / 60)
    if (hour < 24) {
        const restMin = min % 60
        // 余数为 0 就不写，别出现「3 小时 0 分钟」
        return restMin ? `${hour} 小时 ${restMin} 分钟` : `${hour} 小时`
    }
    const day = Math.floor(hour / 24)
    const restHour = hour % 24
    return restHour ? `${day} 天 ${restHour} 小时` : `${day} 天`
}
```

## 3. `src/components/TraceDialog.vue`（新建 · 本项目第一个自写组件）

```vue
<template>
  <el-dialog v-model="visible" title="流转轨迹" width="520">
    <div v-loading="loading" class="trace-body">
      <!-- 顶部摘要：这两样【轨迹接口给不出来】，由父页面传进来 -->
      <el-descriptions v-if="processName || status" :column="1" border class="summary">
        <el-descriptions-item v-if="processName" label="流程">{{ processName }}</el-descriptions-item>
        <el-descriptions-item v-if="status" label="状态">
          <el-tag :type="STATUS_MAP[status].type">{{ STATUS_MAP[status].text }}</el-tag>
        </el-descriptions-item>
      </el-descriptions>

      <el-timeline>
        <el-timeline-item
            v-for="(n, i) in nodes"
            :key="n.activityId + '-' + i"
            :type="nodeType(n, i)"
            :hollow="n.endTime === null"
        >
          <div class="node-title">
            <span class="name">{{ nodeName(n) }}</span>
            <span v-if="n.assigneeName" class="who">{{ n.assigneeName }}</span>
            <el-tag v-if="n.endTime === null" type="primary" size="small">进行中</el-tag>
          </div>
          <div class="node-meta">
            {{ n.startTime }}
            <template v-if="n.endTime">
              → {{ n.endTime }} · 耗时 {{ formatDuration(n.durationInMillis) }}
            </template>
          </div>
        </el-timeline-item>
      </el-timeline>

      <div v-if="!loading && nodes.length === 0" class="empty">暂无轨迹</div>
    </div>

    <template #footer>
      <!-- 只读弹窗：没有提交动作，就不需要 submitting 那一套 -->
      <el-button @click="visible = false">关闭</el-button>
    </template>
  </el-dialog>
</template>

<script setup lang="ts">
import {computed, ref, watch} from 'vue'
import {type ActivityTrace, getActivities} from '@/api/history'
import type {InstanceStatus} from '@/api/instance'
import {formatDuration} from '@/utils/format'

// defineModel() 是 props + emit('update:modelValue') 的语法糖（Vue 3.4+，本项目 3.5.39）。
// 父页面写 <TraceDialog v-model="traceVisible" />，这里直接改 visible.value 就能通知回去。
const visible = defineModel<boolean>({required: true})

const props = defineProps<{
  instanceId: string
  processName?: string | null
  status?: InstanceStatus | null
}>()

const loading = ref(false)
const list = ref<ActivityTrace[]>([])

// 与 myStart 的 STATUS_MAP 是第二份，已登记为 P4 收尾评估项（同后端 nameMap 的第三份重复）
const STATUS_MAP: Record<InstanceStatus, { text: string; type: 'primary' | 'success' | 'danger' }> = {
  RUNNING: {text: '进行中', type: 'primary'},
  COMPLETED: {text: '已办结', type: 'success'},
  TERMINATED: {text: '已终止', type: 'danger'},
}

// activityName 为空时的兜底：BPMN 里事件节点通常不写 name
const TYPE_TEXT: Record<string, string> = {
  startEvent: '开始',
  endEvent: '结束',
  exclusiveGateway: '分支判断',
  parallelGateway: '并行分支',
}

// ⚠️ 必须前端过滤连线：后端故意返回全量（P3 Day 14 决策 2）。
// 不滤的话「开始 → 经理审批 → 结束」三个节点会显示成六七行，一半是箭头。
const nodes = computed(() => list.value.filter(a => a.activityType !== 'sequenceFlow'))

function nodeName(n: ActivityTrace): string {
  return n.activityName || TYPE_TEXT[n.activityType] || n.activityId
}

function nodeType(n: ActivityTrace, i: number): 'primary' | 'success' | 'danger' {
  if (n.endTime === null) {
    return 'primary'          // 当前停留节点，配合 hollow 显示成空心圈
  }
  if (props.status === 'TERMINATED' && i === nodes.value.length - 1) {
    return 'danger'           // 流程在这里被终止（链路没有 endEvent）
  }
  return 'success'
}

async function load() {
  // 先清空：否则第二次打开会先闪一下上一条实例的轨迹
  list.value = []
  if (!props.instanceId) {
    return
  }
  loading.value = true
  try {
    list.value = await getActivities(props.instanceId)
  } catch {
    // 错误提示由 utils/request.ts 的拦截器统一弹
  } finally {
    loading.value = false
  }
}

// 打开时才拉。不能写在 onMounted：组件挂载时弹窗还没开、instanceId 可能是空串
watch(visible, (v) => {
  if (v) {
    load()
  }
})
</script>

<style scoped lang="scss">
.trace-body {
  min-height: 120px;

  .summary {
    margin-bottom: 18px;
  }

  .node-title {
    display: flex;
    align-items: center;
    gap: 8px;

    .name {
      color: var(--el-text-color-primary);
    }

    .who {
      color: var(--el-text-color-regular);
      font-size: 13px;
    }
  }

  .node-meta {
    margin-top: 4px;

  }

  .empty {
    text-align: center;
    padding: 30px 0;
    color: var(--el-text-color-placeholder);
  }
}
</style>
```

    <div class="middle">

          </template>
        </el-table-column>
      </el-table>
    </div>

    <div class="bottom">
      <el-pagination
          class="pager"
          layout="total,prev,pager,next"
          :total="total"
          :current-page="current"
          :page-size="size"
          :disabled="loading"
          @current-change="handlePageChange"
      />
    </div>

    <!-- 已审批行里没有实例状态字段，status 不传（组件里是可选的） -->
    <TraceDialog
        v-model="traceVisible"
        :instance-id="traceRow?.processInstanceId ?? ''"
        :process-name="traceRow?.processDefinitionName ?? null"
    />
  </div>

<script setup lang="ts">
import {onMounted, ref} from 'vue'
import {type ApprovedTask, pageApproved} from '@/api/history'
import {formatDuration} from '@/utils/format'
import TraceDialog from '@/components/TraceDialog.vue'

const current = ref(1)
const size = ref(10)
const total = ref(0)
const list = ref<ApprovedTask[]>([])
const loading = ref(false)

const traceVisible = ref(false)
const traceRow = ref<ApprovedTask | null>(null)

async function loadList() {
  loading.value = true
  try {
    const page = await pageApproved(current.value, size.value)
    list.value = page.records
    total.value = page.total
  } catch {
  } finally {
    loading.value = false
  }
}

function handlePageChange(p: number) {
  current.value = p
  loadList()
}

// 只是打开弹窗（数据由组件自己拉），不发请求，所以不需要 Day 17 那套 openingId 防连点
function openTrace(row: ApprovedTask) {
  traceRow.value = row
  traceVisible.value = true
}

onMounted(() => {
  loadList()
})
</script>

</style>
