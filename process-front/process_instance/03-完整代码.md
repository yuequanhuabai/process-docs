## 1. 新建 `src/api/instance.ts`

```ts
import request from '@/utils/request'
import type { PageResult } from '@/api/user'

// 三种结局：后端按 endTime + deleteReason 判定后给到前端，前端只做展示映射
// 写成联合字面量（而非 string），后面 STATUS_MAP 漏写某个状态时 TS 会报错
export type InstanceStatus = 'RUNNING' | 'COMPLETED' | 'TERMINATED'

// 流程实例：对齐后端改造后的 ProcessInstanceVO
export interface ProcessInstance {
  id: string // 实例 id（引擎 uuid），终止/撤回传它
  processDefinitionId: string // 具体版本的定义 id
  processDefinitionKey: string // 流程 key（跨版本不变），发起时传它
  processDefinitionName: string // 流程名称，如「报销流程」
  businessKey: string | null // 业务标识，可能为空
  startUserId: string // 发起人 userId（数字字符串）
  startUserName: string | null // 发起人姓名 ← 页面显示这个
  startTime: string // 后端已格式化 yyyy-MM-dd HH:mm:ss
  endTime: string | null // 在途为 null
  status: InstanceStatus // 后端算好的结局
  deleteReason: string | null // 终止原因
  currentActivityName: string | null // 当前节点名，已结束为 null
  suspended: boolean // 实例是否挂起（本页不用）
}

export interface StartInstanceBody {
  processDefinitionKey: string
  businessKey?: string
  variables?: Record<string, unknown>
}

// 运行中实例分页（查 RU 表，只有在途的）
// ⚠️ definitionKey 空串要转成 undefined：axios 会丢弃 undefined 的 params，
//    但空串会真的拼出 ?definitionKey= ，后端收到空字符串会过滤出 0 条
export function pageInstances(current = 1, size = 10, definitionKey?: string) {
  return request.get<never, PageResult<ProcessInstance>>('/process-instance/page', {
    params: { current, size, definitionKey: definitionKey || undefined },
  })
}

// 我的发起分页（查 HI 表，在途+办结+终止全都在）
// status 不传 = 全部；同样空串转 undefined
export function myStarted(current = 1, size = 10, status?: string) {
  return request.get<never, PageResult<ProcessInstance>>('/process-instance/my-started', {
    params: { current, size, status: status || undefined },
  })
}

// 发起流程：传 key 不是 id，后端按 key 自动取最新版本
// body 是 post 的第二个参数（不是 { params }，那是 query）
export function startInstance(body: StartInstanceBody) {
  return request.post<never, ProcessInstance>('/process-instance/start', body)
}

// 终止 / 撤回：同一个接口，reason 会写进 DELETE_REASON_（区分办结与终止的判据之一）
// ⚠️ URL 于 2026-08-04 改为 {instanceId}/terminate：原来的 DELETE /process-instance/{id}
//    读起来像"删除这条实例"，而实际语义是"终止"（清 RU 保 HI）。形状与 process-definition
//    的 {definitionId}/suspend、{definitionId}/activate 对齐
export function terminateInstance(instanceId: string, reason?: string) {
  return request.delete<never, void>(`/process-instance/${instanceId}/terminate`, {
    params: { reason: reason || undefined },
  })
}
```

## 2. 新建 `src/views/instance/index.vue`（运行中实例）

```vue
<script setup lang="ts">
import { onMounted, ref } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { pageInstances, terminateInstance, type ProcessInstance } from '@/api/instance'
import { pageDefinitions, type ProcessDefinition } from '@/api/process'

// ---- 列表 + 分页 ----
const list = ref<ProcessInstance[]>([])
const total = ref(0)
const current = ref(1)
const size = ref(10)
const loading = ref(false)

// ---- 流程筛选 ----
const definitionKey = ref('') // '' = 全部流程
const defs = ref<ProcessDefinition[]>([])

async function loadDefs() {
  try {
    // 流程模板就几个，size 给大点一次拉完，不必分页
    const page = await pageDefinitions(1, 100, true)
    defs.value = page.records
  } catch {
    // 已统一弹错
  }
}

async function loadList() {
  loading.value = true
  try {
    const page = await pageInstances(current.value, size.value, definitionKey.value)
    list.value = page.records
    total.value = page.total
  } catch {
    // 错误提示已由 request.ts 拦截器统一处理；
    // 这里只负责别让 rejection 逃出去（调用点没 await，逃出去会变成 unhandled rejection）
  } finally {
    loading.value = false
  }
}

function handlePageChange(p: number) {
  current.value = p
  loadList()
}

// 切换流程筛选：回第一页重查
function handleFilterChange() {
  current.value = 1
  loadList()
}

// ---- 终止 ----
async function handleTerminate(row: ProcessInstance) {
  let reason: string
  try {
    // 用 prompt 不是 confirm —— 后端要收 reason
    const { value } = await ElMessageBox.prompt(
      `确定终止「${row.processDefinitionName}」这条实例？`,
      '终止流程',
      {
        inputPlaceholder: '请输入终止原因，如：填报有误',
        inputValidator: (v: string) => (v && v.trim() ? true : '终止原因不能为空'),
        type: 'warning',
      },
    )
    reason = value
  } catch {
    return // 用户取消（prompt 取消会 reject）
  }
  // 第二个独立的 try/catch：这个管接口失败，别和"用户取消"混在一起
  try {
    await terminateInstance(row.id, reason.trim())
    ElMessage.success('已终止')
    // 终止后该实例从 RU 表消失，若是当前页最后一条则回退一页
    if (list.value.length === 1 && current.value > 1) {
      current.value -= 1
    }
    loadList()
  } catch {
    // 已统一弹错
  }
}

onMounted(() => {
  loadDefs()
  loadList()
})
</script>

<template>
  <div class="instance-page">
    <div class="toolbar">
      <span class="title">运行中实例</span>
      <el-select
        v-model="definitionKey"
        placeholder="全部流程"
        style="width: 180px"
        :disabled="loading"
        @change="handleFilterChange"
      >
        <el-option value="" label="全部流程" />
        <el-option v-for="d in defs" :key="d.key" :value="d.key" :label="d.name" />
      </el-select>
    </div>

    <el-table :data="list" v-loading="loading" border style="width: 100%">
      <el-table-column prop="processDefinitionName" label="流程名称" min-width="120" />
      <el-table-column label="业务标识" min-width="140">
        <template #default="{ row }">{{ row.businessKey || '—' }}</template>
      </el-table-column>
      <el-table-column label="当前节点" min-width="110">
        <template #default="{ row }">{{ row.currentActivityName || '—' }}</template>
      </el-table-column>
      <el-table-column label="发起人" min-width="100">
        <!-- startUserName 万一为空（查不到用户）兜底显示 userId，别留白 -->
        <template #default="{ row }">{{ row.startUserName || row.startUserId }}</template>
      </el-table-column>
      <el-table-column prop="startTime" label="发起时间" width="170" />
      <el-table-column label="操作" width="90" fixed="right">
        <template #default="{ row }">
          <el-button link type="danger" @click="handleTerminate(row)">终止</el-button>
        </template>
      </el-table-column>
    </el-table>

    <el-pagination
      class="pager"
      layout="total, prev, pager, next"
      :total="total"
      :current-page="current"
      :page-size="size"
      :disabled="loading"
      @current-change="handlePageChange"
    />
  </div>
</template>

<style scoped lang="scss">
.instance-page {
  .toolbar {
    display: flex;
    align-items: center;
    justify-content: space-between;
    margin-bottom: 16px;

    .title {
      font-size: 18px;
      font-weight: 600;
      color: #303133;
    }
  }

  .pager {
    margin-top: 16px;
    justify-content: flex-end;
  }
}
</style>
```

## 3. 新建 `src/views/myStart/index.vue`（我的发起 + 发起弹窗）

```vue
<script setup lang="ts">
import { onMounted, reactive, ref } from 'vue'
import { ElMessage, ElMessageBox, type FormInstance, type FormRules } from 'element-plus'
import {
  myStarted,
  startInstance,
  terminateInstance,
  type ProcessInstance,
  type InstanceStatus,
} from '@/api/instance'
import { pageDefinitions, type ProcessDefinition } from '@/api/process'

// 三态映射写成常量对象，别用 v-if 链 —— Day 18 历史页还要再用一次，能直接搬。
// 用 Record<InstanceStatus, ...> 声明：漏写某个状态 TS 会报错
const STATUS_MAP: Record<InstanceStatus, { text: string; type: 'primary' | 'success' | 'danger' }> = {
  RUNNING: { text: '进行中', type: 'primary' },
  COMPLETED: { text: '已办结', type: 'success' },
  TERMINATED: { text: '已终止', type: 'danger' },
}

const statusOptions = [
  { value: '', label: '全部状态' },
  { value: 'RUNNING', label: '进行中' },
  { value: 'COMPLETED', label: '已办结' },
  { value: 'TERMINATED', label: '已终止' },
]

// ---- 列表 + 分页 + 状态筛选 ----
const list = ref<ProcessInstance[]>([])
const total = ref(0)
const current = ref(1)
const size = ref(10)
const loading = ref(false)
const status = ref('') // '' = 全部

async function loadList() {
  loading.value = true
  try {
    // ⚠️ 筛选传给后端，不是前端 list.filter()
    // 前端过滤只能过滤当前这一页，total 还是全量、翻页会出空页
    const page = await myStarted(current.value, size.value, status.value)
    list.value = page.records
    total.value = page.total
  } catch {
    // 已统一弹错
  } finally {
    loading.value = false
  }
}

function handlePageChange(p: number) {
  current.value = p
  loadList()
}

function handleFilterChange() {
  current.value = 1 // 换筛选条件要回第一页，否则可能停在不存在的页码
  loadList()
}

// ---- 发起弹窗 ----
const defs = ref<ProcessDefinition[]>([])
const dialogVisible = ref(false)
const submitting = ref(false)
const formRef = ref<FormInstance>()

const form = reactive({
  processDefinitionKey: '',
  businessKey: '',
  reason: '',
})

const rules: FormRules = {
  processDefinitionKey: [{ required: true, message: '请选择流程', trigger: 'change' }],
}

async function loadDefs() {
  try {
    const page = await pageDefinitions(1, 100, true)
    defs.value = page.records
  } catch {
    // 已统一弹错
  }
}

// 照 views/user/index.vue 的做法：手动重置 + clearValidate。
// 比 resetFields() 可靠——后者依赖每个 el-form-item 都写了 prop，没写的重置不了
function openDialog() {
  form.processDefinitionKey = ''
  form.businessKey = ''
  form.reason = ''
  formRef.value?.clearValidate()
  dialogVisible.value = true
}

async function handleSubmit() {
  const valid = await formRef.value
    ?.validate()
    .then(() => true)
    .catch(() => false)
  if (!valid) return

  submitting.value = true
  try {
    await startInstance({
      processDefinitionKey: form.processDefinitionKey, // 传 key 不是 id
      // 空串转 undefined，别把空字符串存进 businessKey
      businessKey: form.businessKey.trim() || undefined,
      // ⚠️ 业务字段包在 variables 这个 Map 里，不是平铺
      // 变量名是弱契约：引擎不校验拼写，写错会悄悄新建一个变量（P6 表单 schema 要管住这件事）
      variables: form.reason.trim() ? { reason: form.reason.trim() } : undefined,
    })
    ElMessage.success('发起成功')
    dialogVisible.value = false
    current.value = 1
    loadList()
  } catch {
    // 已统一弹错
  } finally {
    submitting.value = false
  }
}

// ---- 撤回（= 终止自己发起的）----
async function handleRevoke(row: ProcessInstance) {
  let reason: string
  try {
    const { value } = await ElMessageBox.prompt(
      `确定撤回「${row.processDefinitionName}」这条申请？`,
      '撤回申请',
      {
        inputPlaceholder: '请输入撤回原因，如：填报有误',
        inputValidator: (v: string) => (v && v.trim() ? true : '撤回原因不能为空'),
        type: 'warning',
      },
    )
    reason = value
  } catch {
    return // 用户取消
  }
  try {
    await terminateInstance(row.id, reason.trim())
    ElMessage.success('已撤回')
    // 注意：这页查 HI 表，撤回后该行不会消失（只是状态变「已终止」），
    // 所以不需要像「运行中实例」页那样回退页码
    loadList()
  } catch {
    // 已统一弹错
  }
}

onMounted(() => {
  loadDefs()
  loadList()
})
</script>

<template>
  <div class="my-start-page">
    <div class="toolbar">
      <span class="title">我的发起</span>
      <div class="toolbar-right">
        <el-select
          v-model="status"
          style="width: 140px"
          :disabled="loading"
          @change="handleFilterChange"
        >
          <el-option v-for="s in statusOptions" :key="s.value" :value="s.value" :label="s.label" />
        </el-select>
        <el-button type="primary" @click="openDialog">发起流程</el-button>
      </div>
    </div>

    <el-table :data="list" v-loading="loading" border style="width: 100%">
      <el-table-column prop="processDefinitionName" label="流程名称" min-width="120" />
      <el-table-column label="业务标识" min-width="140">
        <template #default="{ row }">{{ row.businessKey || '—' }}</template>
      </el-table-column>
      <el-table-column label="状态" width="90">
        <template #default="{ row }">
          <el-tag :type="STATUS_MAP[row.status].type">{{ STATUS_MAP[row.status].text }}</el-tag>
        </template>
      </el-table-column>
      <el-table-column label="当前节点" min-width="110">
        <template #default="{ row }">{{ row.currentActivityName || '—' }}</template>
      </el-table-column>
      <el-table-column prop="startTime" label="发起时间" width="170" />
      <el-table-column label="结束时间" width="170">
        <template #default="{ row }">{{ row.endTime || '—' }}</template>
      </el-table-column>
      <el-table-column label="操作" width="90" fixed="right">
        <template #default="{ row }">
          <!-- 只有进行中的能撤回，已结束的撤不了 -->
          <el-button
            v-if="row.status === 'RUNNING'"
            link
            type="danger"
            @click="handleRevoke(row)"
          >
            撤回
          </el-button>
          <span v-else class="muted">—</span>
        </template>
      </el-table-column>
    </el-table>

    <el-pagination
      class="pager"
      layout="total, prev, pager, next"
      :total="total"
      :current-page="current"
      :page-size="size"
      :disabled="loading"
      @current-change="handlePageChange"
    />

    <el-dialog v-model="dialogVisible" title="发起流程" width="500px">
      <el-form ref="formRef" :model="form" :rules="rules" label-width="92px">
        <el-form-item label="流程" prop="processDefinitionKey">
          <el-select v-model="form.processDefinitionKey" placeholder="请选择" style="width: 100%">
            <el-option v-for="d in defs" :key="d.key" :value="d.key" :label="d.name" />
          </el-select>
        </el-form-item>
        <el-form-item label="业务标识" prop="businessKey">
          <el-input v-model="form.businessKey" placeholder="选填，如 RB-2026-0001" />
        </el-form-item>
        <el-form-item label="申请事由" prop="reason">
          <el-input
            v-model="form.reason"
            type="textarea"
            :rows="3"
            placeholder="选填，会作为流程变量传给审批人查看"
          />
        </el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="dialogVisible = false">取消</el-button>
        <el-button type="primary" :loading="submitting" @click="handleSubmit">确定</el-button>
      </template>
    </el-dialog>
  </div>
</template>

<style scoped lang="scss">
.my-start-page {
  .toolbar {
    display: flex;
    align-items: center;
    justify-content: space-between;
    margin-bottom: 16px;

    .title {
      font-size: 18px;
      font-weight: 600;
      color: #303133;
    }

    .toolbar-right {
      display: flex;
      align-items: center;
      gap: 10px;
    }
  }

  .muted {
    color: #a8abb2;
  }

  .pager {
    margin-top: 16px;
    justify-content: flex-end;
  }
}
</style>
```
