## 1. 新建 `src/api/process.ts`

```ts
import request from '@/utils/request'
import type { PageResult } from '@/api/user'

// 流程定义：对齐后端 ProcessDefinitionVO
// ⚠️ 两个 id 别接反：挂起/激活用 id(=definitionId)，删除用 deploymentId
export interface ProcessDefinition {
  id: string // 流程定义 id（形如 leaveProcess:1:xxx），挂起/激活用它
  key: string // 流程 key（跨版本不变）
  name: string
  version: number
  deploymentId: string // 部署 id，删除用它
  suspended: boolean
  resourceName: string
  diagramResourceName: string | null
}

// 分页列表：latestOnly=true 时每个 key 只返最新版本
export function pageDefinitions(current = 1, size = 10, latestOnly = false) {
  return request.get<never, PageResult<ProcessDefinition>>('/process-definition/page', {
    params: { current, size, latestOnly },
  })
}

// 挂起：禁止该定义再发起新实例（传 definitionId）
export function suspendDefinition(definitionId: string) {
  return request.put<never, void>(`/process-definition/${definitionId}/suspend`)
}

// 激活：恢复发起（传 definitionId）
export function activateDefinition(definitionId: string) {
  return request.put<never, void>(`/process-definition/${definitionId}/activate`)
}

// 删除部署（传 deploymentId）。cascade=false：有在途实例时后端友好报错删不掉；
// cascade=true：连实例带历史一并抹除（本页固定 false，危险操作不默认级联）
export function deleteDeployment(deploymentId: string, cascade = false) {
  return request.delete<never, void>(`/process-definition/deployment/${deploymentId}`, {
    params: { cascade },
  })
}
```

## 2. 新建 `src/views/process/index.vue`

```vue
<script setup lang="ts">
import { onMounted, ref } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import {
  pageDefinitions,
  suspendDefinition,
  activateDefinition,
  deleteDeployment,
  type ProcessDefinition,
} from '@/api/process'

// ---- 列表 + 分页 ----
const list = ref<ProcessDefinition[]>([])
const total = ref(0)
const current = ref(1)
const size = ref(10)
const loading = ref(false)
// 只看每个 key 的最新版本（默认关：同 key 多版本全列出）
const latestOnly = ref(false)

async function loadList() {
  loading.value = true
  try {
    const page = await pageDefinitions(current.value, size.value, latestOnly.value)
    list.value = page.records
    total.value = page.total
  } finally {
    loading.value = false
  }
}

function handlePageChange(p: number) {
  current.value = p
  loadList()
}

// 切换"只看最新版本"：回第一页重查
function handleLatestChange() {
  current.value = 1
  loadList()
}

// ---- 挂起 / 激活（传 definitionId = row.id）----
async function handleToggleSuspend(row: ProcessDefinition) {
  try {
    if (row.suspended) {
      await activateDefinition(row.id)
      ElMessage.success('已激活')
    } else {
      await suspendDefinition(row.id)
      ElMessage.success('已挂起')
    }
    loadList()
  } catch {
    // 错误提示已由 request.ts 拦截器统一处理
  }
}

// ---- 删除部署（传 deploymentId = row.deploymentId，cascade 固定 false）----
async function handleDelete(row: ProcessDefinition) {
  try {
    await ElMessageBox.confirm(
      `确定删除流程「${row.name || row.key}」所在的部署？若该流程下有在途/历史实例将删除失败。`,
      '删除确认',
      { type: 'warning' },
    )
  } catch {
    return // 用户取消
  }
  try {
    await deleteDeployment(row.deploymentId, false)
    ElMessage.success('删除成功')
    // 删掉当前页最后一条时回退一页，避免停在空页
    if (list.value.length === 1 && current.value > 1) {
      current.value -= 1
    }
    loadList()
  } catch {
    // 有在途实例等失败已由拦截器统一弹错
  }
}

onMounted(loadList)
</script>

<template>
  <div class="process-page">
    <div class="toolbar">
      <span class="title">流程管理</span>
      <el-switch
        v-model="latestOnly"
        active-text="只看最新版本"
        @change="handleLatestChange"
      />
    </div>

    <el-table :data="list" v-loading="loading" border style="width: 100%">
      <el-table-column prop="name" label="流程名称" min-width="140" />
      <el-table-column prop="key" label="流程 Key" min-width="140" />
      <el-table-column prop="version" label="版本" width="80" />
      <el-table-column label="状态" width="90">
        <template #default="{ row }">
          <el-tag :type="row.suspended ? 'info' : 'success'">
            {{ row.suspended ? '挂起' : '激活' }}
          </el-tag>
        </template>
      </el-table-column>
      <el-table-column prop="resourceName" label="资源名" min-width="200" show-overflow-tooltip />
      <el-table-column label="操作" width="180" fixed="right">
        <template #default="{ row }">
          <el-button
            link
            :type="row.suspended ? 'success' : 'warning'"
            @click="handleToggleSuspend(row)"
          >
            {{ row.suspended ? '激活' : '挂起' }}
          </el-button>
          <el-button link type="danger" @click="handleDelete(row)">删除</el-button>
        </template>
      </el-table-column>
    </el-table>

    <el-pagination
      class="pager"
      layout="total, prev, pager, next"
      :total="total"
      :current-page="current"
      :page-size="size"
      @current-change="handlePageChange"
    />
  </div>
</template>

<style scoped lang="scss">
.process-page {
  .toolbar {
    display: flex;
    align-items: center;
    justify-content: space-between;
    margin-bottom: 16px;

    .title {
      font-size: 18px;
      font-weight: 600;
      color: #303133;
    }
  }

  .pager {
    margin-top: 16px;
    justify-content: flex-end;
  }
}
</style>
```
