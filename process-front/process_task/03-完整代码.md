## 1. `src/api/task.ts`（新建）

```ts
import request from '@/utils/request'
import type {PageResult} from '@/api/user'

// 待办任务：对齐后端 Day 17 补三字段后的 TaskVO
export interface Task {
    id: string
    name: string
    taskDefinitionKey: string
    assignee: string | null
    processInstanceId: string
    processDefinitionId: string
    createTime: string
    suspended: boolean
    processDefinitionName: string
    businessKey: string | null
    startUserName: string | null
}

// 任务详情：后端 TaskFormVO extends TaskVO，前端照着继承，不平铺重抄
export interface TaskForm extends Task {
    variables: Record<string, unknown>
    formKey: string | null
}

export interface CompleteTaskBody {
    taskId: string
    variables?: Record<string, unknown>
}

// ⚠️ todo 只有 current/size 两个参数，没有筛选——别照 instance 的样子加第三个
export function pageTodo(current = 1, size = 10) {
    return request.get<never, PageResult<Task>>('/task/todo', {
        params: {
            current: current,
            size: size,
        }
    })
}

export function taskDetail(taskId: string) {
    return request.get<never, TaskForm>(`/task/${taskId}`)
}

// ⚠️ taskId 在 body 里，不是 /task/{taskId}/complete（旧文档错过，以 TaskController 为准）
export function completeTask(body: CompleteTaskBody) {
    return request.post<never, void>('/task/complete', body)
}
```

               值是 unknown 只做插值，initiator 显示数字 userId 是已知现状不是 bug -->

<!--所有的基于组件的响应事件处理-->
<script setup lang="ts">
import {computed, onMounted, ref} from "vue";
import {completeTask, pageTodo, type Task, taskDetail, type TaskForm} from "@/api/task";
import {ElMessage} from "element-plus";

const loading = ref(false)
const current = ref(1)
const size = ref(10)
const total = ref(0)
const list = ref<Task[]>([])

// 办理弹窗；没有 formRef/rules——零校验字段（组件选型解析第四节）
const dialogVisible = ref(false)
const submitting = ref(false)
const detail = ref<TaskForm | null>(null)
const comment = ref('')

// 兜住 detail 为 null 的窗口期，模板里 v-for 直接用
const variableEntries = computed(() =>
    detail.value ? Object.entries(detail.value.variables ?? {}) : [])

// 查询待办任务列表数据;
async function loadList() {
  loading.value = true
  try {
    const page = await pageTodo(current.value, size.value)
    list.value = page.records
    total.value = page.total
  } catch {

  } finally {
    loading.value = false
  }
}

// 先查后开：详情拉不到（如"已被他人认领"）就不打开弹窗，错误已由拦截器统一弹
async function openHandle(row: Task) {
  try {
    detail.value = await taskDetail(row.id)
  } catch {
    return
  }
  comment.value = ''
  dialogVisible.value = true
}

async function handleSubmit() {
  if (!detail.value) return
  submitting.value = true
  try {
    // approved 固定 true（决策②：模板无排他网关，"拒绝"按钮是谎言）；
    // comment 空则不塞进 variables，别让引擎存空串变量（决策③）
    const variables: Record<string, unknown> = {approved: true}
    const c = comment.value.trim()
    if (c) {
      variables.comment = c
    }
    await completeTask({taskId: detail.value.id, variables})
    ElMessage.success('已办理')
    // 成功才关弹窗；失败分支不关，用户填的意见还在，可重试或取消
    dialogVisible.value = false
    // 该行要从列表消失：办空末页则回退一页（照实例页 handleTerminate 的三行）
    if (list.value.length === 1 && current.value > 1) {
      current.value -= 1
    }
    loadList()
  } catch {

  } finally {
    submitting.value = false
  }
}

function handlePageChange(p: number) {
  current.value = p
  loadList()
}

onMounted(loadList)
</script>
